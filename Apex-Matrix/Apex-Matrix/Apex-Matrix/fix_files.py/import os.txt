import os
from pathlib import Path

def fix_file_extensions():
    target_dir = Path(".")
    print(f"Scanning working directory: {target_dir.resolve()}")
    
    # Mapping of messy/double-extended names to strict clean filenames
    renames = {
        "matrix.py.py": "matrix.py",
        "matrix.txt": "matrix.py",
        "matrix": "matrix.py",
        "portal.py.txt": "portal.py",
        "portal.txt": "portal.py",
        "portal": "portal.py",
        "requirements.txt.txt": "requirements.txt",
        "requirements": "requirements.txt",
        "main.txt": "main.py",
        "main": "main.py"
    }
    
    for messy, clean in renames.items():
        messy_path = target_dir / messy
        clean_path = target_dir / clean
        
        if messy_path.exists() and messy_path != clean_path:
            if clean_path.exists():
                clean_path.unlink()  # Remove duplicate conflict if necessary
            messy_path.rename(clean_path)
            print(f"[SUCCESS] Renamed '{messy}' -> '{clean}'")
        elif clean_path.exists():
            print(f"[OK] '{clean}' is already correctly named.")
            
    print("\n--- Final Directory Verification ---")
    for f in target_dir.iterdir():
        if f.is_file():
            print(f"📁 {f.name} ({f.stat().st_size} bytes)")

if __name__ == "__main__":
    fix_file_extensions()
